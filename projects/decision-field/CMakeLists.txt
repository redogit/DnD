cmake_minimum_required(VERSION 3.25)
project(decision_field_component_analysis VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(decision_field_component_analysis INTERFACE)
target_include_directories(decision_field_component_analysis INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(decision_field_component_analysis INTERFACE cxx_std_23)
if(MSVC)
  target_compile_options(decision_field_component_analysis INTERFACE /W4 /permissive-)
else()
  target_compile_options(decision_field_component_analysis INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
enable_testing()
function(df_test name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE decision_field_component_analysis)
  add_test(NAME ${name} COMMAND ${name})
endfunction()
df_test(decision_field_component_analysis_tests tests/component_analysis_tests.cpp)
df_test(decision_field_core_coverage_tests tests/core_coverage_tests.cpp)
df_test(decision_field_bridge_tests tests/bridge_tests.cpp)
df_test(decision_field_replay_integrity_tests tests/replay_integrity_tests.cpp)

# Explicit source dependency: never manufacture an engine substitute when absent.
set(DF_PREDECESSOR_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/predecessor/v0_3" CACHE PATH "Verified v0.3 source directory")
if(EXISTS "${DF_PREDECESSOR_SOURCE}/include/decision_field/decision_field.hpp")
  file(SHA256 "${DF_PREDECESSOR_SOURCE}/include/decision_field/decision_field.hpp" engine_hash)
  if(NOT engine_hash STREQUAL "279c4e6cf1ac8b9f9896625bd761d9aa3ed9f89e865c8b47c0061fa28030a34b")
    message(FATAL_ERROR "Unexpected v0.3 engine bytes: preserve as a distinct version instead")
  endif()
  add_subdirectory("${DF_PREDECESSOR_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/predecessor-build")
  df_test(decision_field_predecessor_bridge_tests tests/predecessor_bridge_tests.cpp)
  df_test(decision_field_surface_bridge_tests tests/surface_bridge_tests.cpp)
  foreach(target decision_field_predecessor_bridge_tests decision_field_surface_bridge_tests)
    target_include_directories(${target} PRIVATE "${DF_PREDECESSOR_SOURCE}/include")
  endforeach()
  message(STATUS "Actual v0.3 integration enabled")
else()
  message(STATUS "Standalone bridge tests only; see scripts/recover_predecessor.py for actual v0.3 integration")
endif()

# The immutable predecessor stays intact. Only native-checkpoint targets use the
# hash-bound generated successor; old regression targets still use exact v0.3.
option(DF_ENABLE_NATIVE_PERSISTENCE "Build native checkpoint tests (requires source + OpenSSL)" ON)
if(TARGET decision_field_predecessor_bridge_tests AND DF_ENABLE_NATIVE_PERSISTENCE)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_native_engine.py")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/prepare_native_engine.py"
    "${DF_PREDECESSOR_SOURCE}/include/decision_field/decision_field.hpp" "${CMAKE_CURRENT_BINARY_DIR}/native-generated"
    RESULT_VARIABLE native_prepare_result)
  if(NOT native_prepare_result EQUAL 0)
    message(FATAL_ERROR "Native engine preparation failed")
  endif()
  add_library(decision_field_native_persistence INTERFACE)
  target_link_libraries(decision_field_native_persistence INTERFACE decision_field_component_analysis OpenSSL::Crypto)
  target_include_directories(decision_field_native_persistence BEFORE INTERFACE
    "${CMAKE_CURRENT_BINARY_DIR}/native-generated" "${DF_PREDECESSOR_SOURCE}/include")
  foreach(kind snapshot archive)
    add_executable(decision_field_native_${kind}_tests tests/native_${kind}_tests.cpp)
    target_link_libraries(decision_field_native_${kind}_tests PRIVATE decision_field_native_persistence)
    add_test(NAME decision_field_native_${kind}_tests COMMAND decision_field_native_${kind}_tests)
  endforeach()
  if(UNIX)
    add_executable(decision_field_checkpoint_file_tests tests/checkpoint_file_tests.cpp)
    add_executable(decision_field_checkpoint_process tests/checkpoint_process.cpp)
    foreach(target decision_field_checkpoint_file_tests decision_field_checkpoint_process)
      target_link_libraries(${target} PRIVATE decision_field_native_persistence)
    endforeach()
    add_test(NAME decision_field_checkpoint_file_tests COMMAND decision_field_checkpoint_file_tests "${CMAKE_CURRENT_BINARY_DIR}/checkpoint-file-fixture")
    add_test(NAME decision_field_checkpoint_process_test COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/checkpoint_process_test.py" $<TARGET_FILE:decision_field_checkpoint_process>)
  endif()
endif()
